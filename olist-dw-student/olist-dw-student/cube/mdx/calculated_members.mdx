-- =====================================================================================
-- Olist Sales - calculated members and KPIs (SSAS Multidimensional, MDX script)
-- TP8. Paste this block in the CUBE DESIGNER -> Calculations tab (script view),
-- then Deploy. The cube must have been built first (see cube/README.md).
--
-- Measures referenced here: they MUST have exactly these names in the cube
-- (Measure Group -> Measure Name in the Cube Structure tab):
--   Sales    (fact_sales)    : Sales Amount, Freight Amount, Total Amount,
--                              Item Count, Order Count (distinct count of order_id)
--   Payments (fact_payments) : Payment Value, Installments
--   Orders   (fact_orders)   : Orders Count, Late Orders, Delivery Days, Review Score Sum
--
-- Attributes referenced here MUST exist with these names:
--   [Date].[Calendar]            Year > Quarter > Month > Day  (key = date_key, name = full_date)
--   [Order Status].[Order Status] (key = order_status, name = order_status_label)
--   [Customer].[State] / [Customer].[City] / [Customer].[Customer]
--   [Product].[Category]          (name = category_name_english)
--   [Seller].[Seller]             (name = seller_id)
--   [Payment Type].[Payment Type] (name = payment_type)
--
-- Nothing here changes the numbers: these only add derived KPIs and named sets.
-- =====================================================================================

CALCULATE;

-- --------------------------------------------------------------------------- basic KPIs
CREATE MEMBER CURRENTCUBE.[Measures].[Average Order Value]
 AS IIF([Measures].[Order Count] = 0, NULL,
        [Measures].[Sales Amount] / [Measures].[Order Count]),
 FORMAT_STRING = "Currency", VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Average Item Price]
 AS IIF([Measures].[Item Count] = 0, NULL,
        [Measures].[Sales Amount] / [Measures].[Item Count]),
 FORMAT_STRING = "Currency", VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Freight Percentage]
 AS IIF([Measures].[Sales Amount] = 0, NULL,
        100 * [Measures].[Freight Amount] / [Measures].[Sales Amount]),
 FORMAT_STRING = "0.00", VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Average Freight per Item]
 AS IIF([Measures].[Item Count] = 0, NULL,
        [Measures].[Freight Amount] / [Measures].[Item Count]),
 FORMAT_STRING = "Currency", VISIBLE = 1;

-- ------------------------------------------------------------------- delivery performance
-- Late Rate = Late Orders / DELIVERED Orders. It must be computed on delivered orders only:
-- Orders Count also contains canceled / unavailable / shipped orders, and mixing them in
-- gives 0.79% instead of the expected 6.77%.
-- Inside an MDX expression the measure is reached through its measure group:
-- [Measures].[Late Orders] belongs to the Orders measure group.
CREATE MEMBER CURRENTCUBE.[Measures].[Delivered Orders]
 AS ([Measures].[Orders Count], [Order Status].[Order Status].[Delivered]),
 FORMAT_STRING = "#,#", VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Late Orders Delivered]
 AS ([Measures].[Late Orders], [Order Status].[Order Status].[Delivered]),
 FORMAT_STRING = "#,#", VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Late Rate]
 AS IIF([Measures].[Delivered Orders] = 0, NULL,
        100.0 * [Measures].[Late Orders Delivered] / [Measures].[Delivered Orders]),
 FORMAT_STRING = "0.00\%", VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Late Rate Of All Orders]
 AS IIF([Measures].[Orders Count] = 0, NULL,
        100.0 * [Measures].[Late Orders] / [Measures].[Orders Count]),
 FORMAT_STRING = "0.00\%", VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Average Delivery Days]
 AS IIF([Measures].[Orders Count] = 0, NULL,
        [Measures].[Delivery Days] / [Measures].[Orders Count]),
 FORMAT_STRING = "0.0", VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Average Review Score]
 AS IIF([Measures].[Orders Count] = 0, NULL,
        [Measures].[Review Score Sum] / [Measures].[Orders Count]),
 FORMAT_STRING = "0.00", VISIBLE = 1;

-- ------------------------------------------------------------------------------- payments
CREATE MEMBER CURRENTCUBE.[Measures].[Average Payment Value]
 AS IIF([Measures].[Order Count] = 0, NULL,
        [Measures].[Payment Value] / [Measures].[Order Count]),
 FORMAT_STRING = "Currency", VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Average Installments]
 AS IIF([Measures].[Order Count] = 0, NULL,
        [Measures].[Installments] / [Measures].[Order Count]),
 FORMAT_STRING = "0.00", VISIBLE = 1;

-- ---------------------------------------------------------------------------- time series
-- Same measure computed on the same period of the previous year.
-- The assumption is that the Calendar hierarchy is Year > Quarter > Month > Day:
-- .PARENT.PARENT from a Month lands on the Year, .LAG(1) takes the previous year.
CREATE MEMBER CURRENTCUBE.[Measures].[Sales Amount Previous Year]
 AS ([Measures].[Sales Amount],
     [Date].[Calendar].CURRENTMEMBER.PARENT.PARENT.LAG(1)),
 FORMAT_STRING = "Currency", VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Sales Year over Year Growth]
 AS IIF(([Measures].[Sales Amount],
         [Date].[Calendar].CURRENTMEMBER.PARENT.PARENT.LAG(1)) = 0, NULL,
        ([Measures].[Sales Amount]
         - ([Measures].[Sales Amount], [Date].[Calendar].CURRENTMEMBER.PARENT.PARENT.LAG(1)))
        / ([Measures].[Sales Amount], [Date].[Calendar].CURRENTMEMBER.PARENT.PARENT.LAG(1))),
 FORMAT_STRING = "0.00%", VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Sales Amount Year To Date]
 AS SUM(YTD([Date].[Calendar].CURRENTMEMBER), [Measures].[Sales Amount]),
 FORMAT_STRING = "Currency", VISIBLE = 1;

-- --------------------------------------------------------------------------- named sets
CREATE SET CURRENTCUBE.[Top 10 Categories]
 AS TOPCOUNT([Product].[Category].[Category].MEMBERS, 10, [Measures].[Sales Amount]);

CREATE SET CURRENTCUBE.[Top 10 Sellers]
 AS TOPCOUNT([Seller].[Seller].[Seller].MEMBERS, 10, [Measures].[Sales Amount]);

CREATE SET CURRENTCUBE.[States]
 AS NONEMPTY([Customer].[State].[State].MEMBERS, [Measures].[Sales Amount]);

-- The Calendar Year attribute is built on the `year` column of dim_date, key = 2016.
CREATE SET CURRENTCUBE.[Years 2016-2018]
 AS { [Date].[Calendar].[Calendar Year].&[2016],
      [Date].[Calendar].[Calendar Year].&[2017],
      [Date].[Calendar].[Calendar Year].&[2018] };
